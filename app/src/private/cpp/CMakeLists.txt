cmake_minimum_required(VERSION 3.22)
project(lirix_local_translation LANGUAGES C CXX)
option(LIRIX_PRIVATE "Build the private SentencePiece bridge" OFF)
if(LIRIX_PRIVATE)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_POSITION_INDEPENDENT_CODE ON)
    set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
    set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
    set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
    set(SPM_ENABLE_NFKC_COMPILE OFF CACHE BOOL "" FORCE)
    include(FetchContent)
    FetchContent_Declare(sentencepiece
        URL https://codeload.github.com/google/sentencepiece/tar.gz/refs/tags/v0.2.0
        URL_HASH SHA256=9970f0a0afee1648890293321665e5b2efa04eaec9f1671fcf8048f456f5bb86)
    FetchContent_MakeAvailable(sentencepiece)
    add_library(lirix_sentencepiece SHARED sentencepiece_jni.cpp)
    target_include_directories(lirix_sentencepiece PRIVATE ${sentencepiece_SOURCE_DIR}/src)
    target_link_libraries(lirix_sentencepiece sentencepiece-static)
    if(ANDROID)
        target_link_libraries(lirix_sentencepiece log)
    else()
        find_package(JNI REQUIRED)
        target_include_directories(lirix_sentencepiece PRIVATE ${JNI_INCLUDE_DIRS})
    endif()
endif()
